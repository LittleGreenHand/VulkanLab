cmake_minimum_required(VERSION 3.21)
set(PROJECT_NAME VulkanLab CACHE STRING "可执行文件")
set(AI_PROJECT_NAME VulkanLabAI CACHE STRING "管理AI模型与推理后端的静态库")
project(${PROJECT_NAME})

option(VULKANLAB_BUILD_TESTS "Build VulkanLab unit tests" OFF)

include_directories(thirdParty)
include_directories(thirdParty/glm)
include_directories(thirdParty/gli)
include_directories(thirdParty/imgui)
include_directories(thirdParty/tinygltf)
include_directories(thirdParty/ktx/include)
include_directories(thirdParty/ktx/other_include)

message(STATUS "Using module to find Vulkan:")
find_package(Vulkan)
IF (NOT Vulkan_FOUND)
	message(FATAL_ERROR "Could not find Vulkan library!")
ELSE()
	message(STATUS ${Vulkan_LIBRARY})
ENDIF()

find_package(OpenCV 5 REQUIRED)
IF (NOT OpenCV_FOUND)
	message(FATAL_ERROR "Could not find OpenCV!")
ELSE()
	message(STATUS "OpenCV version: ${OpenCV_VERSION}")
	message(STATUS "OpenCV include direction: ${OpenCV_INCLUDE_DIRS}")
	message(STATUS "OpenCV libs: ${OpenCV_LIBS}")
ENDIF()

if(WIN32)
    add_compile_definitions(NOMINMAX VK_USE_PLATFORM_WIN32_KHR)
    if(MSVC)
        add_compile_options(/utf-8)
    endif()
endif()
add_compile_definitions(_USE_MATH_DEFINES)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}//bin//")
set(SHADERS_OUTPUT "${CMAKE_BINARY_DIR}/bin/shaders" CACHE PATH "shader编译输出目录")

add_subdirectory(shaders)
add_subdirectory(thirdParty/ktx)
add_subdirectory(thirdParty/imgui)

# 不构建 GLFW 自带的 examples / tests / docs
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS    OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS     OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL        OFF CACHE BOOL "" FORCE)
add_subdirectory(thirdParty/glfw-3.5.1)

if(UNIX AND NOT APPLE)
    set(PHYSX_ROOT_DIR "${CMAKE_SOURCE_DIR}/thirdParty/PhysX/physx")
    set(TARGET_BUILD_PLATFORM linux)
    set(PX_GENERATE_GPU_PROJECTS OFF CACHE BOOL "Build PhysX GPU support" FORCE)
    set(PX_GENERATE_GPU_PROJECTS_ONLY OFF CACHE BOOL "Build only PhysX GPU support" FORCE)
    set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "Build static PhysX libraries" FORCE)
    set(PX_BUILDSNIPPETS OFF CACHE BOOL "Build PhysX snippets" FORCE)
    set(PX_BUILDPVDRUNTIME OFF CACHE BOOL "Build PhysX PVD runtime" FORCE)
    set(PX_OUTPUT_LIB_DIR "${CMAKE_BINARY_DIR}" CACHE PATH "PhysX library output directory")
    set(PX_OUTPUT_BIN_DIR "${CMAKE_BINARY_DIR}" CACHE PATH "PhysX binary output directory")
    add_subdirectory("${PHYSX_ROOT_DIR}/compiler/public" physx)
endif()
add_subdirectory(thirdParty/ONNX_Runtime)

add_subdirectory(src/AI)
add_subdirectory(src)

if(VULKANLAB_BUILD_TESTS)
	enable_testing()
	add_subdirectory(tests)
endif()


set_target_properties(glfw PROPERTIES FOLDER "ThirdpartyLib")
set_target_properties(update_mappings PROPERTIES FOLDER "ThirdpartyLib")
set_target_properties(imgui PROPERTIES FOLDER "ThirdpartyLib")
set_target_properties(ktx PROPERTIES FOLDER "ThirdpartyLib")
